; Constants for immediate operands (tools/disasm.py -> src/const.inc)
; NAME = EXPR [@var ...] [; comment]   define; @var (RAM/register name, may be name+N): used automatically where the
;                                      value is stored into var (LDx #v / STx var) or tested against it
;                                      (LDx var / CMP|CPx|AND|ORA|EOR #v, also later CMPs with only branches between)
; use AAAA [AAAA ...] EXPR             the immediate of these instructions is EXPR (checked against the byte)
; EXPR is a ca65 expression: $hex %bin 'c', names, + - * / & | ^ ~ << >> ( ); a leading < or > = low / high byte.
; Definitions may use earlier constants and RAM/register names. (?) = meaning guessed from the code.
; Left as numbers on purpose: #0 / #1 used as clear / flag / step, the negation idiom EOR #$FF + ADC #1,

; ---- PPU

PPUCTRL_INC32 = %00000100 @ppuCtrlShadow ; PPUDATA address +32 (vertical)
PPUCTRL_SPR_1000 = %00001000            ; sprite patterns at $1000
PPUCTRL_BG_1000 = %00010000 @PPUCTRL     ; background patterns at $1000

PPUMASK_SHOW_BG = %00001000
PPUMASK_SHOW_SPR = %00010000
